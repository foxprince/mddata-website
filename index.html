Example: Fetch Latest Quote

REST & WebSocket

JavaScript

import WebSocket from 'ws';

const symbol = 'SZSE:300750';
const ws = new WebSocket('wss://stream.mddata.cc/v1/quotes', {
  headers: { Authorization: `Bearer ${process.env.MDDATA_TOKEN}` }
});

ws.on('open', () => {
  ws.send(JSON.stringify({ action: 'subscribe', channel: 'quotes', symbol }));
});

ws.on('message', (msg) => {
  const data = JSON.parse(msg);
  console.log(`[${data.symbol}] bid ${data.bidPrice} ask ${data.askPrice}`);
});

Python

import requests

API_KEY = 'your-api-key'
symbol = 'AAPL'

resp = requests.get(
    f'https://api.mddata.cc/v1/stock/quote/{symbol}',
    headers={'Authorization': f'Bearer {API_KEY}'}
)
resp.raise_for_status()
payload = resp.json()
print(payload['price'], payload['turnover'])
